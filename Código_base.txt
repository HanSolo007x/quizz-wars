  };

  const app = initializeApp(firebaseConfig);
  const db = getDatabase(app);

  window.db = db;

  window.saveScoreToFirebase = async function(name, points) {
    try {
      console.log("Enviando pontuação para o Realtime Database...");
      const rankingRef = ref(db, "ranking");
      await push(rankingRef, {
        name: name,
        points: Number(points),
        timestamp: serverTimestamp()
      });
      console.log("Pontuação salva com sucesso!");
      await window.loadLeaderboardFromFirebase();
    } catch (e) {
      console.error("Erro ao salvar pontuação no Realtime Database: ", e);
    }
  };

  window.loadLeaderboardFromFirebase = async function() {
    try {
      console.log("Carregando ranking do Realtime Database...");
      const rankingRef = ref(db, "ranking");
      const topScoresQuery = query(rankingRef, orderByChild("points"), limitToLast(10));
      
      const snapshot = await get(topScoresQuery);
      const scores = [];

      if (snapshot.exists()) {
        snapshot.forEach((childSnapshot) => {
          scores.push(childSnapshot.val());
        });
        // Como o limitToLast traz em ordem crescente, invertemos para exibir do maior para o menor
        scores.reverse();
      }

      console.log("Ranking carregado:", scores);
      renderLeaderboardToElement("leaderboard-end", scores);
      renderLeaderboardToElement("leaderboard-standalone", scores);
    } catch (e) {
      console.error("Erro ao buscar ranking do Realtime Database: ", e);
      const errHtml = `<p style='text-align:center; color:#ff4d4d;'>Erro no servidor: ${e.message}</p>`;
      if (document.getElementById("leaderboard-end")) document.getElementById("leaderboard-end").innerHTML = errHtml;
      if (document.getElementById("leaderboard-standalone")) document.getElementById("leaderboard-standalone").innerHTML = errHtml;
    }
  };

  function playSound(type) {
    if (!soundEnabled || !audioCtx) return;

    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);

    const now = audioCtx.currentTime;

    if (type === 'tick') {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(900, now);
      gain.gain.setValueAtTime(0.2, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.05);
      osc.start(now);
      osc.stop(now + 0.05);
    } else if (type === 'click') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(700, now);

  function shuffle(array) {
    let currentIndex = array.length, randomIndex;
    let arrayCopy = [...array];

    while (currentIndex !== 0) {
      randomIndex = Math.floor(Math.random() * currentIndex);
      currentIndex--;
      [arrayCopy[currentIndex], arrayCopy[randomIndex]] = [
        arrayCopy[randomIndex], arrayCopy[currentIndex]
      ];
    }

    return arrayCopy;
  }

  function exitGame() {
    clearInterval(timerInterval);
    showHome();
  }

  function startGame() {
    initAudioContext();
    playSound('click');

    const input = document.getElementById("player-name");
    if (!input.value.trim()) {
      alert("Por favor, insira seu nome!");
      return;
    }
    playerName = input.value.trim();
    
    bgMusic.play().catch(() => console.log("Áudio de fundo bloqueado"));
    
    const q1 = shuffle(questionPool.muitoFacil).slice(0, 4).map(item => ({ ...item, level: 'muitoFacil' }));
    const q2 = shuffle(questionPool.facil).slice(0, 4).map(item => ({ ...item, level: 'facil' }));
    const q3 = shuffle(questionPool.media).slice(0, 4).map(item => ({ ...item, level: 'media' }));
    const q4 = shuffle(questionPool.dificil).slice(0, 4).map(item => ({ ...item, level: 'dificil' }));
    const q5 = shuffle(questionPool.impossivel).slice(0, 4).map(item => ({ ...item, level: 'impossivel' }));

    currentQuestions = [...q1, ...q2, ...q3, ...q4, ...q5];
    
    currentQuestionIndex = 0;
    score = 0;
    correctAnswersCount = 0;

    lifelinesUsed = { fiftyFifty: false, extraTime: false, swap: false };
    document.getElementById("help-5050").disabled = false;
    document.getElementById("help-time").disabled = false;
    document.getElementById("help-swap").disabled = false;

    document.getElementById("start-screen").classList.add("hidden");
    document.getElementById("quiz-screen").classList.remove("hidden");
    showQuestion();
  }

  function toggleAudio() {
    initAudioContext();
    soundEnabled = !soundEnabled;

    if (bgMusic.paused) {
      bgMusic.play();
      muteBtn.innerText = "🔊 Som";
    } else {
      bgMusic.pause();
      muteBtn.innerText = "🔇 Mudo";
    }
  }

  function getTimerForLevel(level) {
    switch (level) {
      case 'muitoFacil': return 10;
      case 'facil': return 15;
      case 'media': return 20;
      case 'dificil': return 25;
      case 'impossivel': return 30;
      default: return 15;
    }
  }

  function startTimer(seconds) {
    clearInterval(timerInterval);
    timeLeft = seconds;
    updateTimerDisplay();

    timerInterval = setInterval(() => {
      timeLeft--;
      updateTimerDisplay();

      if (timeLeft > 0) {
        playSound('tick');
      } else {
        clearInterval(timerInterval);
        triggerFlashEffect('wrong');
        playSound('wrong');
        setTimeout(() => selectAnswer(-1, true), 600);
      }
    }, 1000);
  }

  function updateTimerDisplay() {
    document.getElementById("timer").innerText = `⏱️ ${timeLeft}s`;
  }

  function showQuestion() {
    const q = currentQuestions[currentQuestionIndex];
    document.getElementById("question-number").innerText = `Pergunta ${currentQuestionIndex + 1}/20`;
    document.getElementById("question-text").innerText = q.q;

    document.getElementById("level-display").innerText = `Nível: ${getLevelNameFormatted(q.level)}`;
    document.getElementById("hits-display").innerText = `🎯 Acertos: ${correctAnswersCount}/20`;

    const container = document.getElementById("options-container");
    container.innerHTML = "";

    q.a.forEach((option, index) => {
      const btn = document.createElement("button");
      btn.innerText = option;
      btn.className = "option-btn";
      btn.id = `opt-${index}`;
      btn.onclick = () => handleAnswerClick(index);
      container.appendChild(btn);
    });

    startTimer(getTimerForLevel(q.level));
  }

  function triggerFlashEffect(type) {
    const flashClass = type === 'correct' ? 'flash-correct' : 'flash-wrong';
    mainCard.classList.add(flashClass);
    setTimeout(() => {
      mainCard.classList.remove(flashClass);
    }, 600);
  }

  function handleAnswerClick(index) {
    clearInterval(timerInterval);
    
    const btns = document.querySelectorAll(".option-btn");
    btns.forEach(b => b.disabled = true);

    const isCorrect = (index === currentQuestions[currentQuestionIndex].correct);

    if (isCorrect) {
      playSound('correct');
      triggerFlashEffect('correct');
      score += POINTS_PER_QUESTION;
      correctAnswersCount++;
    } else {
      playSound('wrong');
      triggerFlashEffect('wrong');
    }

    setTimeout(() => {
      selectAnswer(index);
    }, 600);
  }

  function selectAnswer(index, isTimeout = false) {
    currentQuestionIndex++;
    if (currentQuestionIndex < currentQuestions.length) {
      showQuestion();
    } else {
      endGame();
    }
  }

  function use5050() {
    initAudioContext();
    playSound('click');

    if (lifelinesUsed.fiftyFifty) return;
    lifelinesUsed.fiftyFifty = true;
    document.getElementById("help-5050").disabled = true;

    const q = currentQuestions[currentQuestionIndex];
    const correctIdx = q.correct;
    let wrongIndices = [];

    q.a.forEach((_, idx) => {
      if (idx !== correctIdx) wrongIndices.push(idx);
    });

    wrongIndices = shuffle(wrongIndices).slice(0, 2);

    wrongIndices.forEach(idx => {
      const btn = document.getElementById(`opt-${idx}`);
      if (btn) {
        btn.style.visibility = "hidden";
      }
    });
  }

  function useExtraTime() {
    initAudioContext();
    playSound('click');

    if (lifelinesUsed.extraTime) return;
    lifelinesUsed.extraTime = true;
    document.getElementById("help-time").disabled = true;

    timeLeft += 10;
    updateTimerDisplay();
  }

  function useSwap() {
    initAudioContext();
    playSound('click');

    if (lifelinesUsed.swap) return;
    
    const currentQ = currentQuestions[currentQuestionIndex];
    const levelPool = questionPool[currentQ.level];

    const available = levelPool.filter(item => 
      !currentQuestions.some(cq => cq.q === item.q)
    );

    if (available.length === 0) {
      alert("Não há outras perguntas disponíveis para troca neste nível.");
      return;
    }

    lifelinesUsed.swap = true;
    document.getElementById("help-swap").disabled = true;

    const newQ = shuffle(available)[0];
    currentQuestions[currentQuestionIndex] = { ...newQ, level: currentQ.level };
    showQuestion();
  }

  function endGame() {
    clearInterval(timerInterval);
    document.getElementById("quiz-screen").classList.add("hidden");
    document.getElementById("end-screen").classList.remove("hidden");
    document.getElementById("final-score").innerText = score;

    if (window.saveScoreToFirebase) {
      window.saveScoreToFirebase(playerName, score);
    }
  }

  function getMedalIcon(index) {
    switch (index) {
      case 0: return '🥇 ';
      case 1: return '🥈 ';
      case 2: return '🥉 ';
      default: return `${index + 1}º `;
    }
  }

  function renderLeaderboardToElement(elementId, scores) {
    const list = document.getElementById(elementId);
    if (!list) return;
    list.innerHTML = "";

    if (scores.length === 0) {
      list.innerHTML = "<p style='text-align:center; color:#888;'>Nenhuma pontuação registrada.</p>";
      return;
    }

    scores.forEach((s, idx) => {
      const li = document.createElement("li");
      li.className = `rank-${idx}`;
      li.innerHTML = `
        <span><strong>${getMedalIcon(idx)}</strong>${s.name}</span>
        <span style="color: #FFE81F; font-weight: bold;">${s.points} pts</span>
      `;
      list.appendChild(li);
    });
  }
